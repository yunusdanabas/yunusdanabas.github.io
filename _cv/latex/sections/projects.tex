\section*{PROJECTS}

\textbf{Hand-Steer Sim --- Gesture Teleoperation for Mobile Robots} \textbar\ \cvlink{https://yunusemredanabas.com/projects/hand_steer_sim/}{\texttt{project page}} \textbar\ \cvlink{https://github.com/yunusdanabas/hand_steer_sim}{\texttt{code}} \hfill Mar 2025--May 2025
\begin{itemize}
  \item Developed a camera-only teleoperation stack for differential-drive robots in ROS Noetic: MediaPipe hand landmarks $\rightarrow$ dual-branch TFLite inference (MLP for static commands, LSTM for steering trajectories) $\rightarrow$ \texttt{geometry\_msgs/Twist}
  \item Achieved 99.65\% accuracy (static) and 99.77\% macro-F1 (dynamic) on a held-out test set; measured 13 ms end-to-end latency on an RTX 4060 Ti ($\sim$25 ms on laptop CPU) at 30 FPS
  \item Packaged a reproducible ROS+Gazebo workspace with one-command \texttt{roslaunch}; released a data-recording GUI, training notebooks, pretrained TFLite models, and CPU/GPU Dockerfiles
\end{itemize}
